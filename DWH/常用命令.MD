# 查看自动刷新任务历史（核心！看有没有定时跑、成功与否）
SELECT scheduled_start_time,start_time,end_time,state,process_id
FROM information_schema.mv_auto_refresh_jobs
WHERE mv_schema='dwd' AND mv_name='dwd_usr_app_user'
ORDER BY scheduled_start_time DESC LIMIT 20;

# state 状态：
# PENDING：等待调度
# RUNNING：正在刷新
# FINISHED：✅ 刷新成功
# FAILED：❌ 刷新失败（会自动重试最多 10 次）
# TIMEOUT：超时异常
# 表最多保留 1024 条刷新记录；创建 MV 时的首次全量加载不会进这张表，只有 NEXT 自动周期的任务才会记录。

# 手动刷新
REFRESH MATERIALIZED VIEW dwd.dwd_usr_app_user;

# 物化视图删除
